    MSB                     LSB
# instruções com apenas registradores
    ADD:  0101_ddd_sss_sss_111 -- ADD destino, fonte1(3 ao 5), fonte2(6 ao 8)
    SUB:  0101_ddd_sss_sss_000 -- SUB destino, fonte1(3 ao 5), fonte2(6 ao 8)
    XOR:  0101_ddd_sss_sss_010 -- XOR destino, fonte1(3 ao 5), fonte2(6 ao 8)
    AND:  0101_ddd_sss_sss_011 -- AND destino, fonte1(3 ao 5), fonte2(6 ao 8)
    #fonte1 e destino sempre são sempre o acumulador das operações acima

    MOV:  0101_ddd_sss_xxx_001 -- MOV destino, fonte(6 ao 8) 

# instruções envolvendo imediato
    SUBI: 0110_ddd_sss_cccccc  -- SUBI destino, fonte(6 ao 8), constante    # os opcodes são diferentes
    CMPI: 1001_sss_ccccccccc   -- CMPI registrador (acumulador), constante  # para manter um maior intervalo
    LD:   0010_ddd_ccccccccc   -- LD destino, constante                     # para as constantes
    # registrador do CMPI, destino e fonte do SUBI são sempre o acumulador

# instrução de salto incondicional
    JUMP: 1111_xxxxx_ccccccc   -- JUMP constante

# instrução de nop
    NOP:  0000_xxxxxxxxxxxx    -- NOP

# instruções de salto condicional
    BEQ:  1100_cccccccc_x_010 -- BEQ delta
    BGE:  1100_cccccccc_x_101 -- BGE delta

# instruções de memória
    LW:   0011_ddd_aaa_xxxxxx -- LW destino, endereço # o destino de dados vem do banco de registradores
    SW:   1101_aaa_sss_xxxxxx -- SW fonte, endereço   # a fonte de dados vem do banco de registradores
    # o endereço é sempre o dado advindo do acumulador


# detalhes
opcode - 4 bits MSB
func   - 3 bits MSB, quando há (instruções apenas com registradores e branches)

x: bits que não importam
sss: registrador fonte
ddd: registrador destino
c: representa bits de constante
aaa: reg que diz o endereco
