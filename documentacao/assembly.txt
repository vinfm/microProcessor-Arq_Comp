-- Assembly para a validação --

--Preencher RAM com números ate 62 -
00 - LD A, 2           -- Acumulador A recebe 2 (primeiro número a ser armazenado)
01 - LD R1, 1          -- R1 = 1 (incrementador para os loops)
02 - LD R3, 2          -- R3 = 2 (endereço inicial da RAM para armazenar os números)
03 - MOV R0, A         -- R0 recebe o valor atual de A
04 - SW R0, R3         -- RAM[R3] = R0 (armazena o valor de R0 no endereço R3)
05 - ADD A, A, R1      -- A = A + 1 (incrementa o valor a ser armazenado)
06 - ADD R3, R3, R1    -- R3 = R3 + 1 (incrementa o endereço de armazenamento)
07 - CMPI A, 63        -- Compara A com 63 (verifica se chegou ao fim)
08 - BGE 2             -- Se A >= 63, pula 2 linhas (vai para linha 10, fim do preenchimento)
09 - JUMP 3            -- Volta para linha 03 (continua preenchendo)

-- Começa a eliminação de múltiplos de 2 --
10 - LD R2, 2          -- R2 = 2 (divisor atual)
11 - LD R3, 4          -- R3 = 4 (primeiro múltiplo de 2 a eliminar)
12 - MOV A, R3         -- A = R3 (A recebe o endereço do múltiplo)
13 - LW R0, A          -- R0 = RAM[A] (carrega valor da RAM no A)
14 - MOV R4, R0        -- R4 = R0 (copia valor para R4)
15 - MOV R5, R4        -- R5 = R4 (salva valor antes da subtração)
16 - SUB R4, R4, R2    -- R4 = R4 - R2 (subtrai divisor até R4 < 0)
17 - CMPI R4, 0        -- Compara R4 com 0
18 - BGE -3            -- Se R4 >= 0, volta para MOV R5, R4 (linha 15)
19 - CMPI R5, 0        -- Compara valor salvo antes da última subtração
20 - BEQ 2             -- Se R5 == 0, pula para linha 22 (vai zerar o múltiplo)
21 - JUMP 23           -- Senão, pula para próximo endereço
22 - LD R0, 0          -- R0 = 0 (prepara para zerar RAM)
23 - SW R0, A          -- RAM[A] = 0 (zera múltiplo)
24 - ADD R3, R3, R1    -- R3 = R3 + 1 (próximo endereço)
25 - CMPI R3, 63       -- Compara R3 com 63 (fim da RAM?)
26 - BGE 2             -- Se R3 >= 63, pula para linha 28 (fim da eliminação de múltiplos de 2)
27 - JUMP 12           -- Volta para linha 12 (continua eliminando)
