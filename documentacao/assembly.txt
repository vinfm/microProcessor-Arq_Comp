--Preencher RAM com números ate 62 -
00 - LD R3, 0       -- 0010_100_000000000
01 - LD R1, 1       -- 0010_001_000000001
02 - MV A, R3       -- 0101_111_011_010_001
03 - LW R3, A       -- 0011_011_111_000000
04 - ADD A, A, R1   -- 0101_111_001_111_111
05 - CMPI A, 128    -- 1001_111_010000000
06 - BGE 2          -- 1100_00000010_1_101
07 - JUMP 3         -- 1111_11111_0000011


-- Eliminação eficiente de múltiplos de 2 --
08 - LD R2, 2          -- R2 = 2 (divisor)
09 - LD A, 4           -- R3 = 4 (primeiro múltiplo de 2)
10 - LD R0, 0          -- R0 = 0 (valor para zerar)
11 - SW R0, A          -- RAM[A] = 0 (zera múltiplo)
12 - ADD A, A, R2      -- R3 = R3 + R2 (vai para o próximo múltiplo)
13 - CMPI A, 63        -- Compara A com 63
14 - BGE 2             -- Se A >= 63, pula para o próximo bloco
15 - JUMP 10           -- Volta para linha 10 (continua zerando múltiplos)
