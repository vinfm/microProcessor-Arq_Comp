-- Assembly para a validação --

--Preencher RAM com números ate 62 -
00 - LD A, 2           -- Acumulador A recebe 2 (primeiro número a ser armazenado)
01 - LD R1, 1          -- R1 = 1 (incrementador para os loops)
02 - LD R3, 2          -- R3 = 2 (endereço inicial da RAM para armazenar os números)
03 - MOV R0, A         -- R0 recebe o valor atual de A
04 - SW R0, R3         -- RAM[R3] = R0 (armazena o valor de R0 no endereço R3)
05 - ADD A, A, R1      -- A = A + 1 (incrementa o valor a ser armazenado)
06 - ADD R3, R3, R1    -- R3 = R3 + 1 (incrementa o endereço de armazenamento)
07 - CMPI A, 63        -- Compara A com 63 (verifica se chegou ao fim)
08 - BGE 2             -- Se A >= 63, pula 2 linhas (vai para linha 10, fim do preenchimento)
09 - JUMP 3            -- Volta para linha 03 (continua preenchendo)

-- Eliminação eficiente de múltiplos de 2 --
10 - LD R2, 2          -- R2 = 2 (divisor)
11 - LD R3, 4          -- R3 = 4 (primeiro múltiplo de 2)
12 - MOV A, R3         -- A = R3
13 - LD R0, 0          -- R0 = 0 (valor para zerar)
14 - SW R0, A          -- RAM[A] = 0 (zera múltiplo)
15 - ADD R3, R3, R2    -- R3 = R3 + R2 (vai para o próximo múltiplo)
16 - CMPI R3, 63       -- Compara R3 com 63
17 - BGE 2             -- Se R3 >= 63, pula para o próximo bloco
18 - JUMP 12           -- Volta para linha 12 (continua zerando múltiplos)

-- Eliminação eficiente de múltiplos de 3 --

27 - JUMP 21           -- Volta para linha 21

-- Eliminação eficiente de múltiplos de 5 --

-- Varredura da saída --
46 - LD R3, 2          -- R3 = 2 (endereço inicial para varredura)
47 - MOV A, R3         -- A = R3
